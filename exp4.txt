import tensorflow as tf
from tensorflow.keras import layers, models
import matplotlib.pyplot as plt

(x_train, _), (x_test, _) = tf.keras.datasets.mnist.load_data()

x_train = x_train.astype("float32").reshape(-1, 28, 28, 1) / 255
x_test = x_test.astype("float32").reshape(-1, 28, 28, 1) / 255

inp = tf.keras.Input((28, 28, 1))

x = layers.Conv2D(16, 3, activation='relu', padding='same')(inp)
x = layers.MaxPooling2D(2, padding='same')(x)
x = layers.Conv2D(8, 3, activation='relu', padding='same')(x)
x = layers.MaxPooling2D(2, padding='same')(x)

x = layers.Conv2D(8, 3, activation='relu', padding='same')(x)
x = layers.UpSampling2D(2)(x)
x = layers.Conv2D(16, 3, activation='relu', padding='same')(x)
x = layers.UpSampling2D(2)(x)

out = layers.Conv2D(1, 3, activation='sigmoid', padding='same')(x)

autoencoder = models.Model(inp, out)

autoencoder.compile(
    optimizer='adam',
    loss='binary_crossentropy'
)

autoencoder.fit(
    x_train,
    x_train,
    epochs=10,
    batch_size=128,
    shuffle=True,
    validation_data=(x_test, x_test)
)

decoded = autoencoder.predict(x_test, verbose=0)

plt.figure(figsize=(20, 4))

for i in range(10):
    plt.subplot(2, 10, i + 1)
    plt.imshow(x_test[i].reshape(28, 28), cmap='Blues')
    plt.axis('off')

    plt.subplot(2, 10, i + 11)
    plt.imshow(decoded[i].reshape(28, 28), cmap='Blues')
    plt.axis('off')

plt.show()
