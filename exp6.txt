import numpy as np
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import mean_absolute_error, mean_squared_error
from tensorflow.keras.models import Model
from tensorflow.keras.layers import Input, LSTM, Dense, TimeDistributed

t = np.arange(1000)
series = 10 + 0.01*t + np.sin(2*np.pi*t/50) + 0.1*np.random.randn(1000)
scaler = StandardScaler()
data = scaler.fit_transform(series.reshape(-1, 1)).flatten()

IW, OW = 30, 10
X = np.array([data[i:i+IW] for i in range(len(data)-IW-OW)])
Y = np.array([data[i+IW:i+IW+OW] for i in range(len(data)-IW-OW)])
X, Y = X[..., None], Y[..., None]

s = int(0.8*len(X))
Xtr, Xte, Ytr, Yte = X[:s], X[s:], Y[:s], Y[s:]

ei = Input((IW, 1))
_, h, c = LSTM(64, return_state=True)(ei)
di = Input((OW, 1))
d = LSTM(64, return_sequences=True)(di, initial_state=[h, c])
out = TimeDistributed(Dense(1))(d)

model = Model([ei, di], out)
model.compile('adam', 'mse', metrics=['mae'])
model.fit([Xtr, np.zeros_like(Ytr)], Ytr, epochs=20, batch_size=32,
          validation_split=0.1, verbose=1)

pred = model.predict([Xte, np.zeros_like(Yte)])
pred = scaler.inverse_transform(pred.reshape(-1, 1)).flatten()
actual = scaler.inverse_transform(Yte.reshape(-1, 1)).flatten()

print("MAE:", mean_absolute_error(actual, pred))
print("RMSE:", np.sqrt(mean_squared_error(actual, pred)))
